decl
  int swap(int *p, int *q);
  int *identity(int *p);
enddecl

int swap(int *p, int *q)
{
  decl
    int temp;
  enddecl

  begin
    temp=*p;
    *p=*q;
    *q=temp;
    return 0;
  end
}

int *identity(int *p)
{
  begin
    return p;
  end
}

int main()
{
  decl
    int a,b;
    int *p;
  enddecl

  begin
    read(a);
    read(b);
    p=identity(&a);
    swap(p,&b);
    write(a);
    write(b);
    return 0;
  end
}
