class
    Point{
        decl
            int x;
            int y;
        enddecl
    }
endclass

int calculate(int n){
    decl
        Point p;
        int result;
    enddecl
    begin
        p = new(Point);
        p.x = n;
        p.y = n + 1;
        result = p.x + p.y;
        return result;
    end
}

int main(){
    decl
        int result;
    enddecl
    begin
        result = Initialize();
        result = calculate(10);
        write(result);
        return 0;
    end
}
